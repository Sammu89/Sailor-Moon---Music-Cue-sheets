# Helper function to process time field for OCR error correction
def process_time_field(time_str):
    time_str = time_str.replace("′", "min").replace("″", "sec").replace("°", "º")
    time_str = time_str.replace("半", ".5").replace("'", "min").replace("minmin", "min")
    if re.match(r"^\d+(\.\d+)?$", time_str):
        time_str += "sec"
    return time_str
